  // Look closer: works like the puzzle room. One finger moves the picture, two fingers pinch to zoom

  function minS(){ return Math.min(1, sc.clientWidth / (SW || 1), sc.clientHeight / (SH || 1)); }
  function clampS(v){ return Math.min(MAXS, Math.max(minS(), v)); }
  // Never lose the picture: it can't be pulled off an edge, and when it's smaller than the screen it sits in the middle.
  function clamp(){
    var W = sc.clientWidth, H = sc.clientHeight; Z.s = clampS(Z.s);
    var w = SW * Z.s, h = SH * Z.s;
    Z.x = w <= W ? (W - w) / 2 : Math.min(0, Math.max(W - w, Z.x));
    Z.y = h <= H ? (H - h) / 2 : Math.min(0, Math.max(H - h, Z.y));
  }

  function up(e){
    var had = ptrs[e.pointerId]; delete ptrs[e.pointerId]; var n = Object.keys(ptrs).length;
    if (pinch) {
