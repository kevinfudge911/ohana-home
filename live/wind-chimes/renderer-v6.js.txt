const resolveStyle=style=>{if(typeof style!=='string'||!Object.hasOwn(STYLES,style))throw new RangeError('Unsupported V6 chime style');return STYLES[style];};
const NS='http://www.w3.org/2000/svg';
const node=(tag,attrs={})=>{const e=document.createElementNS(NS,tag);for(const[k,v]of Object.entries(attrs))e.setAttribute(k,String(v));return e;};
const image=(name,x,y,w,h,aspect='xMidYMid meet')=>node('image',{href:'/wind-chimes/art/'+name,x,y,width:w,height:h,preserveAspectRatio:aspect});
const text=(content,x,y,size,attrs={})=>{const e=node('text',{x,y,'text-anchor':'middle','font-family':'Georgia, serif','font-size':size,fill:'#eee8d9',...attrs});e.textContent=content;return e;};
const path=(d,stroke='#28251f',width=1.7)=>node('path',{d,stroke,'stroke-width':width,fill:'none','stroke-linecap':'round'});
const clamp=(v,max)=>Math.max(-max,Math.min(max,Number(v)||0));
const rotation=(a,x,y)=>`rotate(${(a*180/Math.PI).toFixed(5)} ${x} ${y})`;
let serial=0;
export function rotatePoint(p,at,a){const c=Math.cos(a),s=Math.sin(a),x=p[0]-at[0],y=p[1]-at[1];return[at[0]+x*c-y*s,at[1]+x*s+y*c];}
// A connector and its child share one endpoint and the same transform hierarchy.
// The exported geometry is also used by the deterministic continuity tests.
export function buildGeometry(style='porch-primary'){
 const def=resolveStyle(style),family=def.family;
 const geometry={family,width:family==='primary'?250:family==='umbrella'?400:500,height:family==='primary'?1000:family==='umbrella'?950:1100,parts:[]};
 if(family==='primary'){
  const xs=[56,91,127,163,197];
  const lengths=[420,461,480,445,402],ys=[252,264,251,267,260];
  geometry.parts=def.order.map((part,i)=>({part,pivot:[xs[i],198],loop:[xs[i],ys[part]+4],top:ys[part],width:42,height:lengths[part],depth:[1,3].includes(i)?'back':'front'}));
  geometry.center={pivot:[125,198],upperEnd:[125,500],clapperCenter:[125,511],lowerStart:[125,525],sailLoop:[125,823],sailWidth:def.memorial?72:46,sailHeight:def.memorial?110:128};
 }else if(family==='umbrella'){
  // Mounts lie ON the measured opaque lower metal rim of the 333x512 ornament sprite.
  const rings=[[80,479],[120,497],[165,503],[210,496],[250,477]],indices=def.count===3?[0,2,4]:def.count===4?[0,1,3,4]:[0,1,2,3,4],scale=360/333;
  const source=[[195,423,97,26],[149,329,74,28],[162,315,82,26],[182,364,91,27],[125,313,62,27]];
  const extension=def.count===3?[48,150,48]:def.count===4?[48,136,136,48]:[38,105,150,105,38];
  geometry.parts=def.order.map((part,i)=>{const ring=rings[indices[i]],pivot=[20+ring[0]*scale,ring[1]*scale],h=[208,180,216,195,190][part],s=h/source[part][1];return{part,pivot,loop:[pivot[0],pivot[1]+extension[i]],source:source[part],scale:s,width:source[part][0]*s,height:h,extension:extension[i]};});
 }else{
  // Existing crystal ornament/artwork retained. Visible ring centers, not generic X positions.
  const rings=[[51,457],[147,450],[251,449],[364,450],[460,454]],indices=def.count===2?[1,3]:def.count===3?[0,2,4]:def.count===4?[0,1,3,4]:[0,1,2,3,4],scale=450/512;
  const source=[[59,512,30,16],[66,512,33,16],[70,512,35,15],[62,512,31,16],[62,512,31,15]],hs=[463,510,620,500,553];
  geometry.parts=def.order.map((part,i)=>{const ring=rings[indices[i]],pivot=[25+ring[0]*scale,ring[1]*scale],s=hs[part]/512;return{part,pivot,loop:[...pivot],source:source[part],scale:s,width:source[part][0]*s,height:hs[part],extension:0};});
 }
 return geometry;
}
function chain(g,x,y0,y1){
 // Touching alternating elliptical links form a continuous chain, including both endpoint eyes.
 const length=y1-y0;if(length<=0)return;
 g.append(path(`M${x} ${y0}L${x} ${y1}`,'#493b23',1.1));
 const count=Math.max(1,Math.ceil(length/5)),step=length/count;
 for(let i=0;i<=count;i++)g.append(node('ellipse',{cx:x,cy:y0+i*step,rx:i%2?1.25:2.3,ry:step*.6,fill:'none',stroke:i%2?'#6d582e':'#ab8c46','stroke-width':1.05}));
}
export function mountRig(host,style='porch-primary'){
 const def=resolveStyle(style),geo=buildGeometry(style),W=geo.width,H=geo.height,id='v6-rig-'+(++serial);
 const svg=node('svg',{viewBox:`0 0 ${W} ${H}`,'aria-hidden':'true',focusable:'false',class:'ohana-layered-chime','data-articulation':'connected-v1'});Object.assign(svg.style,{width:'100%',height:'100%',overflow:'visible',pointerEvents:'none'});
 const assembly=node('g',{'data-part':'assembly'});svg.append(assembly);const moving=[];let upper=null,lower=null;
 if(geo.family==='primary'){
  const defs=node('defs'),wood=node('linearGradient',{id:id+'-wood',x1:'0%',y1:'0%',x2:'0%',y2:'100%'});
  wood.append(node('stop',{offset:'0%','stop-color':'#343431'}),node('stop',{offset:'60%','stop-color':'#181918'}),node('stop',{offset:'100%','stop-color':'#41403a'}));defs.append(wood);svg.append(defs);
  const rear=node('g',{'data-depth':'rear-tubes'}),center=node('g',{'data-depth':'center-pendulum'}),front=node('g',{'data-depth':'front-tubes'});
  assembly.append(rear,center,front);
  geo.parts.forEach((p,i)=>{
   const g=node('g',{'data-part':'tube-'+i,'data-pivot':p.pivot.join(',')});
   g.append(path(`M${p.pivot[0]} ${p.pivot[1]}L${p.loop[0]} ${p.loop[1]}`));
   // Generated cardinal/floral atlas is isolated with native SVG viewports; original reference photos are never published.
    const boxes=[[89,24,118,1126],[356,45,117,1081],[620,63,117,1063],[881,89,115,1040],[1151,70,114,1058]],box=boxes[p.part];
    const sprite=node('svg',{x:p.loop[0]-p.width/2,y:p.top,width:p.width,height:p.height,viewBox:box.join(' '),preserveAspectRatio:'none',overflow:'hidden','data-texture':'cardinal-floral-reference'});
    sprite.append(node('image',{href:'/wind-chimes/art/v6-atlas.png',x:0,y:0,width:1347,height:1168,preserveAspectRatio:'none'}));g.append(sprite);
   (p.depth==='back'?rear:front).append(g);moving.push({g,p});
  });
  const c=geo.center;upper=node('g',{'data-part':'center-upper','data-pivot':c.pivot.join(',')});
  upper.append(path(`M${c.pivot[0]} ${c.pivot[1]}L${c.upperEnd[0]} ${c.upperEnd[1]}`,'#211f1b',1.7));
  upper.append(node('path',{d:'M83 503Q125 495 167 503L167 517Q125 534 83 517Z',fill:`url(#${id}-wood)`,stroke:'#161715','stroke-width':1}),node('ellipse',{cx:125,cy:503,rx:42,ry:9,fill:'#454540',stroke:'#20211d','stroke-width':1}));
  // Both lower line and sail rotate about the clapper underside INSIDE the upper pendulum.
  // This nested transform keeps the lower cord physically attached for every pose.
  lower=node('g',{'data-part':'center-lower','data-pivot':c.lowerStart.join(',')});
  lower.append(path(`M${c.lowerStart[0]} ${c.lowerStart[1]}L${c.sailLoop[0]} ${c.sailLoop[1]}`,'#211f1b',1.7));
  if(def.memorial){lower.append(node('path',{d:'M91 835Q91 823 125 823Q159 823 159 835L161 925Q125 938 89 925Z',fill:'#282b28',stroke:'#4e514a','stroke-width':1.2}),text('In Memory',125,858,10,{'font-style':'italic'}),text('of',125,873,9,{'font-style':'italic'}),text('Daddy',125,896,17));}
  else{lower.append(node('path',{d:'M125 823C119 850 101 898 102 924C102 958 148 958 148 924C149 898 131 850 125 823Z',fill:'#691e2e',stroke:'#37111d','stroke-width':1.1}),node('path',{d:'M125 834C120 858 106 900 107 923',fill:'none',stroke:'#a84859','stroke-width':1.2,opacity:'.55'}));}
  lower.append(node('ellipse',{cx:125,cy:823,rx:2,ry:3,fill:'none',stroke:'#8c8071','stroke-width':1.2}));upper.append(lower);center.append(upper);
  // Cover suspension starts with the canopy lip rather than leaving floating line ends.
  assembly.append(image('primary-canopy.webp',0,0,250,268*250/329));
 }else if(geo.family==='umbrella'){
  assembly.append(image('umbrella-ornament.webp',20,0,360,512*360/333));
  for(const p of geo.parts){const g=node('g',{'data-part':'bell-'+p.part,'data-pivot':p.pivot.join(',')});chain(g,p.pivot[0],p.pivot[1],p.loop[1]);g.append(image('umbrella-strand'+(p.part+1)+'.webp',p.loop[0]-p.source[2]*p.scale,p.loop[1]-p.source[3]*p.scale,p.width,p.height));assembly.append(g);moving.push({g,p});}
 }else{
  assembly.append(image('crystal-ornament.webp',25,0,450,472*450/512));
  for(const p of geo.parts){const g=node('g',{'data-part':'crystal-'+p.part,'data-pivot':p.pivot.join(',')});g.append(image('crystal-strand'+(p.part+1)+'.webp',p.loop[0]-p.source[2]*p.scale,p.loop[1]-p.source[3]*p.scale,p.width,p.height));assembly.append(g);moving.push({g,p});}
 }
 host.replaceChildren(svg);host.dataset.chimeStyle=style;host.style.pointerEvents='none';let last='';
 return {svg,partCount:moving.length,get bounds(){return{width:W,height:H}},geometry:geo,
  paint(r){const angles=(r.tubes||[]).map(v=>clamp(v,.11)*(geo.family==='primary'?.38:.55)),a=clamp(r.angle,.085)*.35,s=clamp(r.striker,.115)*.28,l=clamp(r.sail,.14)*.22;
   const key=[a,...angles,s,l].map(v=>v.toFixed(6)).join(',');if(key===last)return;last=key;
   assembly.setAttribute('transform',rotation(a,W/2,8));moving.forEach((m,i)=>m.g.setAttribute('transform',rotation(angles[i]||0,...m.p.pivot)));
   upper?.setAttribute('transform',rotation(s,...geo.center.pivot));lower?.setAttribute('transform',rotation(l,...geo.center.lowerStart));
  },destroy(){svg.remove()}};
}
