import {ChimeController} from './core-layered.js';
import {audioHostContext,createHostAudioGuard} from './host-audio-contract.js';
import {installChimeSoundControls} from './chime-sound-controls.js';

// The renderer and target are supplied by the app; invalid scenes park the host.
export function createPreparedChimeHost({readState,readScene=()=>null,getContext,getGlobalMuted=()=>true,getBusy=()=>false,
 document:doc=globalThis.document,events=globalThis,storage=globalThis.localStorage,
 loadAudio=()=>Promise.all([import('./recorded-audio-layered.js'),import('./audio/catalog.js')]),
 loadRenderer=null,getRenderTarget=()=>null,loadWeather=null,setInterval:setTimer=(f,t)=>globalThis.setInterval(f,t),clearInterval:clearTimer=id=>globalThis.clearInterval(id),
 requestFrame=callback=>globalThis.requestAnimationFrame(callback),cancelFrame=id=>globalThis.cancelAnimationFrame(id)}={}){
 if(typeof readState!=='function')throw Error('Sanitized host reader required');
 let destroyed=false,suspended=false,generation=0,frame=null,mounted=null,pendingMount=null,audio=null,controls=null;
 let audioModules=null,audioPromise=null,rendererPromise=null,previousRoute=null,previousSession=null,previousActivity=null,previousSoundScope=null,bridgeSoundScope=null,bridgeReturnActivity=null,bridgeReturnPending=null,failedRoute=null,geometryFailure=null,geometryRetries=0;
 const MAX_GEOMETRY_RETRIES=6,nodeIds=new WeakMap();let nextNodeId=0;
 const stats={audioImports:0,rendererImports:0,mounts:0,staleMounts:0,paints:0,geometryUpdates:0,geometryRetries:0};
 const state=()=>{try{return readState()||{};}catch{return{};}};
 const muted=()=>{try{return!!getGlobalMuted()||!!controls?.isMuted();}catch{return true;}};
 const busy=()=>{try{return!!getBusy();}catch{return true;}};
 const current=()=>{const value=state();return{value,context:audioHostContext(value)};};
 const controller=new ChimeController({storage,allowSharedWeather:typeof loadWeather==='function'});
 // Real local wind: the app supplies loadWeather (an authenticated /api/wind read). The chimes swing and
 // strike with the actual wind at the viewer's location; when no reading is available they keep the
 // simulated breeze. A reading is reapplied whenever the controller falls back to simulated wind.
 let weather=null,weatherTimer=null,weatherBusy=false;const WEATHER_FRESH=1800000,WEATHER_POLL=600000;
 const weatherFresh=()=>!!weather&&Date.now()-weather.updatedAt<WEATHER_FRESH;
 function applyWeather(){if(destroyed||!weatherFresh()||controller.wind.mode==='shared-weather')return;controller.setSharedWeather(weather);}
 async function refreshWeather(){if(destroyed||typeof loadWeather!=='function'||weatherBusy)return;const value=state();if(value.authenticated!==true||value.visible!==true)return;weatherBusy=true;try{const w=await loadWeather();if(!destroyed&&w&&w.available===true&&w.units==='m/s'&&Number.isFinite(w.speedMps)){weather={authorized:true,available:true,units:'m/s',speedMps:w.speedMps,gustMps:Number.isFinite(w.gustMps)?w.gustMps:w.speedMps,directionDeg:Number.isFinite(w.directionDeg)?w.directionDeg:0,updatedAt:Date.now(),place:w.place||null};controller.setSharedWeather(weather);}}catch{}finally{weatherBusy=false;}}
 if(typeof loadWeather==='function'){controller.subscribe(s=>{if(s.mode==='simulated'&&s.session)applyWeather();});weatherTimer=setTimer(()=>{if(!weatherFresh()||Date.now()-weather.updatedAt>WEATHER_POLL)refreshWeather();else applyWeather();},60000);Promise.resolve().then(refreshWeather);}
 const valid=(stamp,key)=>!destroyed&&!suspended&&generation===stamp&&current().context.routeKey===key&&current().context.visible;
 function cancelOwnedFrame(){if(frame!==null){cancelFrame(frame);frame=null;}}
 function releaseMounted(){cancelOwnedFrame();const old=mounted;mounted=null;if(old){try{old.instance?.destroy?.();}catch{}old.layer.remove();}controller.setScene('',[]);}
 function park(reason,{preserveSound=false}={}){generation++;pendingMount=null;if(!preserveSound){bridgeReturnPending=null;controller.mute();}controller.audio?.stop(reason);releaseMounted();}
 // Bridge tab changes are controls navigation, not a new room or account.
 const soundScope=(value,context)=>JSON.stringify([context.sessionKey,...['roomId','roomEpoch','gameId','gameType','gameEntryVisit','puzzleId','recordId','pageId'].map(k=>value[k]??null),value.visible===true,value.eligible===true,value.appInert===true]);
 function audioEligible(){const {value,context}=current();return!destroyed&&!suspended&&context.visible&&context.routeKey===previousRoute&&context.sessionKey===controller.sessionKey&&value.paused!==true&&value.reducedMotion!==true&&controller.active;}
 const guard=createHostAudioGuard({controller,readState:()=>{const value=state();return suspended?{...value,visible:false}:value;}});
 function ensureAudio(){
  if(audio||destroyed||suspended)return;
  if(!audioPromise){stats.audioImports++;audioPromise=Promise.resolve().then(loadAudio).then(modules=>{audioModules=modules;return modules;});audioPromise.catch(()=>{});}
  const stamp=generation,key=current().context.routeKey;
  audioPromise.then(([engine,catalog])=>{
   if(audio||!valid(stamp,key))return;
   // The catalog remains unreviewed. The owner explicitly authorized live listening,
   // which is a separate eligibility flag and still needs an intentional trusted tap.
   audio=new engine.RecordedChimes({clips:catalog.clips,qualityReviewed:catalog.qualityReviewed,ownerLiveTestAllowed:true,
    getContext,isMuted:muted,isBusy:busy,isActive:audioEligible,document:doc,events});
   controller.audio=audio;controller.audioStatus=audio.available?'Sound off':'Recording awaiting listening review';
   audio.setVolume(controller.prefs.volume);controller.notify();restoreChosenSound();controls?.refresh();
  }).catch(()=>{if(valid(stamp,key)){controller.audioStatus='Sound unavailable; remains silent';controller.notify();}});
 }
 function resolveTarget(value){try{return getRenderTarget(value)||null;}catch{return null;}}
 // A route/modal park stops voices, not the user's saved speaker choice. Re-arm only
 // an already decoded recording in this still-running page context; browser unlock
 // after a reload or suspended context always remains a trusted speaker tap.
 function restoreChosenSound(){if(controls?.wantsSound()&&!muted()&&!doc.hidden)controller.resumeSoundIfUnlocked();}
 function safeScene(){try{return readScene()||null;}catch{return null;}}
 function nodeId(node){if(!node||typeof node!=='object')return null;if(!nodeIds.has(node))nodeIds.set(node,++nextNodeId);return nodeIds.get(node);}
 // A failed placement may be retried only after an actual scene/geometry change, never on every poll or frame.
 function geometryFingerprint(){
  const s=safeScene(),win=doc?.defaultView;
  if(!s)return 'scene-unavailable';
  const element=e=>{if(!e||typeof e.getBoundingClientRect!=='function')return[nodeId(e),false];
   try{const r=e.getBoundingClientRect(),css=win?.getComputedStyle(e),host=e.getRootNode?.()?.host,art=host?.images?.island;
    return[nodeId(e),e.isConnected===true,r.x,r.y,r.width,r.height,css?.display,css?.visibility,css?.opacity,css?.transform,css?.backgroundImage,css?.backgroundSize,css?.backgroundPosition,css?.objectFit,css?.objectPosition,e.currentSrc||e.src||null,e.complete??null,e.naturalWidth??null,e.naturalHeight??null,e.width??null,e.height??null,host?.artUrl||null,art?.naturalWidth??null,art?.naturalHeight??null];}
   catch{return[nodeId(e),'unreadable'];}};
  return JSON.stringify([s.sceneId,s.routeId,s.styleId,s.accountEpoch,s.routeEpoch,s.sceneEpoch,s.modalOpen,s.musicActive,
   win?.visualViewport?.offsetLeft,win?.visualViewport?.offsetTop,win?.visualViewport?.width,win?.visualViewport?.height,
   element(s.target),element(s.sourceElement),Array.isArray(s.obstacles)?s.obstacles.map(element):null]);
 }
 function geometryScope(){const s=safeScene();return s?JSON.stringify([s.sceneId,s.sceneEpoch,nodeId(s.target),nodeId(s.sourceElement)]):'scene-unavailable';}
 function rejectRenderer(instance,key,reason){
  if(instance?.failureKind==='geometry'){
   const scope=geometryScope(),accepted=mounted?.instance===instance?mounted.acceptedScope:null;
   // A source/scene replacement invalidates the old mount but may already be valid for a new one.
   // Grant exactly one event-driven remount even when the post-rejection geometry is stable.
   geometryFailure={key,sessionKey:current().context.sessionKey,signature:geometryFingerprint(),scope,retries:geometryRetries,scopeReplacement:accepted!==null&&accepted!==scope};
  }
  else{failedRoute=key;geometryFailure=null;}
  park(reason);
 }
 function validMountedSpecs(specs){
  if(!Array.isArray(specs)||specs.length===0)return false;
  const ids=new Set();
  for(const spec of specs){
   if(!spec||typeof spec!=='object'||Array.isArray(spec)||typeof spec.id!=='string'||!spec.id.trim()||
    /[\u0000-\u001f\u007f]/.test(spec.id)||!Number.isInteger(spec.count)||spec.count<1||spec.count>7||ids.has(spec.id))return false;
   ids.add(spec.id);
  }
  return true;
 }
 function safeRendererState(){const {value,context}=current();return{value,context,target:value.rendererEligible===true?resolveTarget(value):null};}
 function wantsRenderer(){const {value,context,target}=safeRendererState();return!destroyed&&!suspended&&context.visible&&value.activity!=='bridge'&&value.rendererEligible===true&&target?.isConnected===true;}
 function scheduleFrame(){if(frame===null&&mounted&&controller.needsFrame&&wantsRenderer())frame=requestFrame(paint);}
 function paint(time){
  frame=null;if(destroyed||suspended||!mounted)return;
  const {context,target}=safeRendererState();
  if(context.routeKey!==mounted.key||target!==mounted.target||!wantsRenderer()){refresh();return;}
  try{controller.step(time);const painted=mounted.instance.paint?.(controller.snapshot);if(painted===false||mounted.instance.active===false){rejectRenderer(mounted.instance,context.routeKey,'renderer-inactive');return;}mounted.acceptedScope=geometryScope();stats.paints++;scheduleFrame();}
  catch{rejectRenderer(mounted.instance,context.routeKey,'renderer-paint-failed');}
 }
 function ensureRenderer(){
  if(typeof loadRenderer!=='function'||!wantsRenderer())return;
  const {context,target}=safeRendererState(),key=context.routeKey,stamp=generation;
  if(failedRoute===key)return;
  if(geometryFailure?.key===key){
   if(geometryFailure.sessionKey!==context.sessionKey){geometryFailure=null;geometryRetries=0;}
   else{const signature=geometryFingerprint(),newScope=geometryScope()!==geometryFailure.scope,acceptedReplacement=geometryFailure.scopeReplacement===true;
    if((!acceptedReplacement&&signature===geometryFailure.signature)||
     (geometryFailure.retries>=MAX_GEOMETRY_RETRIES&&!newScope&&!acceptedReplacement))return;
    geometryRetries=newScope?0:geometryFailure.retries+1;stats.geometryRetries++;geometryFailure=null;}
  }
  if(mounted&&mounted.key===key&&mounted.target===target){try{const updated=mounted.instance.updateGeometry?.();if(updated===false||mounted.instance.active===false){rejectRenderer(mounted.instance,key,'renderer-inactive');return;}stats.geometryUpdates++;if(mounted.reduced!==controller.reduced){mounted.reduced=controller.reduced;const painted=mounted.instance.paint?.(controller.snapshot);if(painted===false||mounted.instance.active===false){rejectRenderer(mounted.instance,key,'renderer-inactive');return;}stats.paints++;}mounted.acceptedScope=geometryScope();}catch{rejectRenderer(mounted.instance,key,'renderer-geometry-failed');return;}if(!controller.needsFrame)cancelOwnedFrame();scheduleFrame();return;}
  if(pendingMount?.stamp===stamp&&pendingMount.target===target)return;
  if(!rendererPromise){stats.rendererImports++;rendererPromise=Promise.resolve().then(loadRenderer);rendererPromise.catch(()=>{});}
  const pending={stamp,target};pendingMount=pending;
  rendererPromise.then(async module=>{
   if(!valid(stamp,key)||!wantsRenderer()||resolveTarget(state())!==target)return;
   if(typeof module.mount!=='function')throw Error('Verified renderer adapter required');
   const layer=doc.createElement('div');layer.dataset.chimePreparation='';layer.setAttribute('aria-hidden','true');layer.inert=true;
   layer.style.cssText='position:absolute;inset:0;pointer-events:none;overflow:hidden';
   let instance;
   try{instance=await module.mount({layer,context:{...context,readScene:safeScene},mode:'simulated'});}
   catch(error){layer.remove();throw error;}
   if(!valid(stamp,key)||!wantsRenderer()||resolveTarget(state())!==target){stats.staleMounts++;try{instance?.destroy?.();}catch{}layer.remove();return;}
   // Only anchors accepted by this mounted adapter may become controller rigs.
   const specs=instance?.specs;
   if(instance?.active===false||!validMountedSpecs(specs)){rejectRenderer(instance,key,'renderer-inactive');try{instance?.destroy?.();}catch{}layer.remove();return;}
   releaseMounted();controller.setScene(key,specs);mounted={key,target,layer,instance:instance||{},reduced:controller.reduced,acceptedScope:geometryScope()};target.append(layer);stats.mounts++;
   try{const painted=mounted.instance.paint?.(controller.snapshot);if(painted===false||mounted.instance.active===false){rejectRenderer(mounted.instance,key,'renderer-inactive');return;}mounted.acceptedScope=geometryScope();stats.paints++;geometryFailure=null;geometryRetries=0;bridgeReturnPending=null;}catch{rejectRenderer(mounted.instance,key,'renderer-paint-failed');return;}
   scheduleFrame();
  }).catch(()=>{if(valid(stamp,key))rejectRenderer(null,key,'renderer-load-failed');}).finally(()=>{if(pendingMount===pending)pendingMount=null;});
 }
 function refresh(){
  if(destroyed)return;
  const {value,context}=current(),routeChanged=previousRoute!==null&&previousRoute!==context.routeKey,scope=soundScope(value,context);
  if(value.activity==='bridge'&&previousActivity!=='bridge'){bridgeSoundScope=scope;bridgeReturnActivity=previousActivity??'room';}
  const bridgeTransition=value.activity==='bridge'&&previousActivity!==null&&previousActivity!=='bridge'&&previousSoundScope===scope||
   previousActivity==='bridge'&&bridgeSoundScope===scope&&(value.activity==='bridge'||value.activity===bridgeReturnActivity);
  // Only an already completed trusted enable may survive a same-session Bridge round trip.
  // Voices stop and rigs leave during the modal; Music, hidden, pause, mute and account changes still cancel intent.
  const preserveSound=bridgeTransition&&controller.sound===true&&previousSession===context.sessionKey&&context.visible&&value.reducedMotion!==true&&!suspended&&!muted();
  if(routeChanged){park('host-route-change',{preserveSound});bridgeReturnPending=preserveSound&&value.activity!=='bridge'?scope:null;failedRoute=null;geometryFailure=null;geometryRetries=0;}
  if(previousSession!==null&&previousSession!==context.sessionKey){audio?.destroy();audio=null;controller.audio=null;}
  previousSession=context.sessionKey;previousRoute=context.routeKey;previousActivity=value.activity;previousSoundScope=scope;if(value.activity!=='bridge'){bridgeSoundScope=null;bridgeReturnActivity=null;}
  guard.refresh();
  const paused=value.paused===true;if(controller.prefs.paused!==paused){controller.prefs.paused=paused;controller.mute();controller.reset('host-pause');}
  const controlsVisible=context.sessionKey&&value.visible===true&&value.eligible===true&&value.appInert!==true&&value.activity!=='music'&&!suspended;
  if(controlsVisible&&!controls)controls=installChimeSoundControls({controller,document:doc,events,storage,getGlobalMuted,onTrustedEnable:()=>{const next=current();if(next.value.activity==='bridge'&&next.context.sessionKey===controller.sessionKey&&next.context.visible&&controller.sound&&!muted())bridgeSoundScope=soundScope(next.value,next.context);}});
  if(!controlsVisible&&controls){controls.destroy();controls=null;}
  controller.globalMuteChanged();restoreChosenSound();controls?.refresh();
  if(!context.visible||suspended){controller.mute();releaseMounted();return;}
  ensureAudio();
  if(!wantsRenderer()){if(mounted||pendingMount)park('renderer-target-unavailable',{preserveSound:bridgeReturnPending===scope&&controller.sound===true&&previousSession===context.sessionKey&&context.visible&&value.rendererEligible===true&&value.reducedMotion!==true&&!muted()});else releaseMounted();return;}
  ensureRenderer();
 }
 function suspend(reason='host-hidden'){if(destroyed)return;suspended=true;park(reason);guard.stop(reason);controls?.destroy();controls=null;}
 function resume(){if(destroyed)return;const wasSuspended=suspended;suspended=false;if(wasSuspended){geometryFailure=null;geometryRetries=0;}refresh();}
 function destroy(){if(destroyed)return;suspend('host-destroyed');destroyed=true;if(weatherTimer!==null){clearTimer(weatherTimer);weatherTimer=null;}controls?.destroy();controls=null;guard.destroy();controller.destroy();audio=null;}
 return{refresh,resume,suspend,destroy,controller,get snapshot(){return{phase:'V6-INTEGRATION-CANDIDATE',weather:weather?{...weather}:null,windMode:controller.wind.mode,rendererInstalled:typeof loadRenderer==='function',rendererStatus:mounted?'v6-mounted':typeof loadRenderer==='function'?'v6-pending':'v6-unavailable',audioLoaded:!!audio,audioAvailable:audio?.available===true,generation,framesPending:frame!==null,mounted:!!mounted,recoveryPending:!!geometryFailure,stats:{...stats},destroyed};}};
}
