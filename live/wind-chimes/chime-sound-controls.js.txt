// Bind to the existing chime controller and shared host context; never make an engine.
export const CHIME_SOUND_KEY='ohana_chime_sounds_v1';
export function installChimeSoundControls({controller,document:doc=globalThis.document,events=globalThis,storage=globalThis.localStorage,getGlobalMuted=()=>true}={}){
 if(!controller||typeof controller.enableSound!=='function')throw Error('Existing chime controller required');
 const clamp=v=>Math.max(0,Math.min(1,v));let prefs={volume:clamp((controller.prefs?.volume??.18)/.5),muted:true},persisted=true,destroyed=false,mounted=null,queued=false;const listeners=[];
 function read(){try{const p=JSON.parse(storage.getItem(CHIME_SOUND_KEY)||'{}');if(typeof p.volume==='number'&&Number.isFinite(p.volume))prefs.volume=clamp(p.volume);if(typeof p.muted==='boolean')prefs.muted=p.muted;}catch{}}
 read();const masterMuted=()=>{try{return!!getGlobalMuted();}catch{return true;}};
 const originalEnable=controller.enableSound;const boundEnable=originalEnable.bind(controller);
 function save(){try{storage.setItem(CHIME_SOUND_KEY,JSON.stringify(prefs));persisted=true;}catch{persisted=false;}}
 const isMuted=()=>prefs.muted||prefs.volume===0||masterMuted();
 function setVolume(v){if(destroyed||typeof v!=='number'||!Number.isFinite(v))return;prefs.volume=clamp(v);save();controller.setVolume(prefs.volume*.5);refresh();}
 function mute(){if(destroyed)return;prefs.muted=true;save();controller.mute();refresh();}
 async function enable(event){if(destroyed||!event?.isTrusted)return false;prefs.muted=false;save();controller.setVolume(prefs.volume*.5);if(doc.hidden||masterMuted()||!controller.active){refresh();return false;}const ok=await boundEnable(event);if(!destroyed)refresh();return ok;}
 // Existing Settings Enable remains deliberate and uses this independent preference.
 controller.enableSound=enable;controller.setVolume(prefs.volume*.5);controller.mute();
 function render(){if(!mounted)return;const volume=Math.round(prefs.volume*100),input=mounted.panel.querySelector('input'),button=mounted.panel.querySelector('[data-chime-mute]');if(doc.activeElement!==input)input.value=String(volume);input.setAttribute('aria-valuetext',volume+'%');mounted.panel.querySelector('output').textContent=volume+'%';const muted=prefs.muted,enabled=controller.sound;button.textContent=muted?'Unmute':'Mute';button.setAttribute('aria-label',muted?'Unmute wind chime sounds':'Mute wind chime sounds');button.setAttribute('aria-pressed',String(muted));mounted.panel.querySelector('[data-chime-enable]').hidden=muted||enabled||!controller.audio?.available;mounted.panel.querySelector('small').textContent=!persisted?'For this visit; storage is unavailable.':masterMuted()?'Game sound is muted. Chime settings are saved.':!controller.audio?.available?'Recording unavailable; sounds stay off.':controller.prefs.paused?'Chimes are paused. Sound setting saved.':controller.reduced?'Reduced motion: automatic strikes are paused.':enabled?'Live listening test enabled.':'Live listening test: enable with a tap for this visit.';}
 function unmount(){if(!mounted)return;mounted.panel.removeEventListener('input',mounted.input);mounted.panel.removeEventListener('click',mounted.click);mounted.panel.remove();mounted=null;}
 function refresh(){if(destroyed)return;const dialog=doc.getElementById('ohana-chat-dialog'),target=dialog?.open&&controller.session&&!doc.hidden?dialog.querySelector('.uc-command-sections'):null;if(mounted&&(!target||mounted.dialog!==dialog||!target.contains(mounted.panel)))unmount();if(target&&!mounted){const panel=doc.createElement('section');panel.className='ohana-chime-sound-controls';panel.dataset.chimeProtect='';panel.setAttribute('aria-label','Wind chime sound controls');panel.innerHTML='<h3>Wind chime sounds</h3><p>Separate settings on this device. Game sound mute still silences chimes.</p><div><label for="ohana-chime-volume">Wind chime volume</label><output for="ohana-chime-volume"></output><input id="ohana-chime-volume" aria-label="Wind chime volume" type="range" min="0" max="100" step="1"><button type="button" data-chime-mute>Unmute</button><button type="button" data-chime-enable hidden>Enable for this visit</button><small></small></div>';const input=e=>{if(e.target.matches('input'))setVolume(Number(e.target.value)/100);},click=e=>{if(e.target.closest('[data-chime-enable]'))void enable(e);else if(e.target.closest('[data-chime-mute]')){if(!prefs.muted)mute();else void enable(e);}};panel.addEventListener('input',input);panel.addEventListener('click',click);target.append(panel);mounted={dialog,panel,input,click};}render();}
 const schedule=()=>{if(destroyed||queued)return;queued=true;queueMicrotask(()=>{queued=false;refresh();});};
 const observer=new MutationObserver(records=>{const relevant=records.some(r=>{if(r.type==='attributes')return r.target.id==='ohana-chat-dialog';if(r.target.closest?.('[data-daddy-audio-controls],.ohana-chime-sound-controls'))return false;if(mounted?.dialog.contains(r.target))return true;return[...r.addedNodes,...r.removedNodes].some(n=>n.nodeType===1&&(n.id==='ohana-chat-dialog'||n.querySelector?.('#ohana-chat-dialog')));});if(relevant)schedule();});observer.observe(doc.body,{childList:true,subtree:true,attributes:true,attributeFilter:['open']});
 const unsubscribe=controller.subscribe(s=>{const v=s.volume/.5;if(prefs.volume!==v){prefs.volume=v;save();}render();});
 function on(target,type,fn){target.addEventListener(type,fn);listeners.push(()=>target.removeEventListener(type,fn));}
 on(doc,'visibilitychange',()=>{if(doc.hidden)controller.mute();refresh();});on(events,'pagehide',()=>{controller.mute();unmount();});on(events,'storage',e=>{if(e.key===CHIME_SOUND_KEY||e.key===null){read();controller.setVolume(prefs.volume*.5);controller.mute();}if(e.key==='ohana_muted'||e.key===null)controller.globalMuteChanged();refresh();});on(doc,'click',()=>{controller.globalMuteChanged();render();});
 const style=doc.createElement('style');style.textContent='#ohana-chat-dialog .ohana-chime-sound-controls{display:block!important;max-width:100%;min-width:0;padding:12px;background:#fff5db;color:#173b35;border:1px solid #aa854f;border-radius:14px;margin:12px 0;text-shadow:none}.ohana-chime-sound-controls h3{font-size:18px}.ohana-chime-sound-controls p,.ohana-chime-sound-controls small{font-size:12px;line-height:1.45}.ohana-chime-sound-controls>div{display:grid;grid-template-columns:minmax(0,1fr) auto;gap:4px 8px}.ohana-chime-sound-controls input{grid-column:1;min-width:0;width:100%;min-height:44px}.ohana-chime-sound-controls button{grid-column:2;min-height:44px;min-width:70px}.ohana-chime-sound-controls small{grid-column:1/-1}.ohana-chime-sound-controls [data-chime-enable]{grid-column:1/-1}.ohana-chime-sound-controls [hidden]{display:none!important}';doc.head.append(style);refresh();
 return{isMuted,setVolume,mute,enable,refresh,get snapshot(){return{...prefs,enabled:controller.sound,available:controller.audio?.available===true,persisted,mounted:!!mounted};},destroy(){if(destroyed)return;destroyed=true;controller.mute();observer.disconnect();unsubscribe();listeners.splice(0).forEach(f=>f());if(controller.enableSound===enable)controller.enableSound=originalEnable;unmount();style.remove();}};
}
