// Bind to the existing chime controller and shared host context; never make an engine.
export const CHIME_SOUND_KEY='ohana_chime_sounds_v1';
export function installChimeSoundControls({controller,document:doc=globalThis.document,events=globalThis,storage=globalThis.localStorage,getGlobalMuted=()=>true,onTrustedEnable=()=>{}}={}){
 if(!controller||typeof controller.enableSound!=='function')throw Error('Existing chime controller required');
 const clamp=v=>Math.max(0,Math.min(1,v));let prefs={volume:clamp((controller.prefs?.volume??.18)/.5),muted:true},persisted=true,destroyed=false,mounted=null,queued=false;const listeners=[];
 function read(){prefs.muted=true;try{const p=JSON.parse(storage.getItem(CHIME_SOUND_KEY)||'{}');if(typeof p.volume==='number'&&Number.isFinite(p.volume))prefs.volume=clamp(p.volume);if(typeof p.muted==='boolean')prefs.muted=p.muted;}catch{}}
 read();const masterMuted=()=>{try{return!!getGlobalMuted();}catch{return true;}};
 const originalEnable=controller.enableSound;const boundEnable=originalEnable.bind(controller);
 function save(){try{storage.setItem(CHIME_SOUND_KEY,JSON.stringify(prefs));persisted=true;}catch{persisted=false;}}
 const isMuted=()=>prefs.muted||prefs.volume===0||masterMuted();
 function setVolume(v){if(destroyed||typeof v!=='number'||!Number.isFinite(v))return;prefs.volume=clamp(v);save();controller.setVolume(prefs.volume*.5);refresh();}
 function mute(){if(destroyed)return;prefs.muted=true;save();controller.mute();refresh();}
 async function enable(event){if(destroyed||!event?.isTrusted)return false;prefs.muted=false;save();controller.setVolume(prefs.volume*.5);if(doc.hidden||masterMuted()||!controller.active){refresh();return false;}const ok=await boundEnable(event);if(!destroyed&&ok&&controller.sound&&!prefs.muted){try{onTrustedEnable();}catch{}}if(!destroyed)refresh();return ok;}
 // The speaker is the only enable control. Mounting the panel must not erase a
 // previously unlocked same-session ON choice.
 controller.enableSound=enable;controller.setVolume(prefs.volume*.5);if(prefs.muted)controller.mute();
 function render(){
  if(!mounted)return;
  const volume=Math.round(prefs.volume*100),input=mounted.panel.querySelector('input'),button=mounted.panel.querySelector('[data-chime-mute]'),status=mounted.panel.querySelector('[data-chime-status]');
  if(doc.activeElement!==input)input.value=String(volume);
  input.setAttribute('aria-valuetext',volume+'%');mounted.panel.querySelector('output').textContent=volume+'%';
  const muted=prefs.muted,globalMuted=masterMuted(),available=controller.audio?.available===true,paused=controller.prefs.paused,reduced=controller.reduced;
  const contextState=controller.audio?.ctx?.state,unlocked=contextState==='running'&&!!controller.audio?.buffers?.length;
  const recordingPending=!!controller.audio?.pending,recordingFailed=controller.audio?.reason==='decode-failed'||/unavailable/i.test(controller.audioStatus||'');
  const playing=!muted&&!globalMuted&&prefs.volume>0&&available&&!paused&&!reduced&&controller.sound&&unlocked;
  const message=muted?'SOUND OFF':globalMuted?'SOUND ON — GAME SOUND MUTED':prefs.volume===0?'SOUND ON — VOLUME 0':!available?'SOUND ON — RECORDING UNAVAILABLE':paused?'SOUND ON — CHIMES PAUSED':reduced?'SOUND ON — REDUCED MOTION':playing?'SOUND ON':!contextState?'SOUND ON — BROWSER TAP NEEDED':contextState!=='running'?'SOUND ON — BROWSER AUDIO PAUSED':recordingPending?'SOUND ON — RECORDING LOADING':recordingFailed?'SOUND ON — RECORDING UNAVAILABLE':!unlocked?'SOUND ON — RECORDING NOT READY':'SOUND ON — PLAYBACK PAUSED';
  button.dataset.silent=String(muted);button.setAttribute('aria-label',muted?'Turn wind chime sounds on':'Turn wind chime sounds off');button.setAttribute('aria-pressed',String(!muted));
  status.dataset.tone=muted?'muted':playing?'enabled':'off';if(status.textContent!==message)status.textContent=message;
  mounted.panel.querySelector('small').textContent=!persisted?'Choice lasts for this visit; storage is unavailable.':globalMuted?'Game sound is muted. Your chime choice is saved.':!available||recordingFailed?'The recording is unavailable. Your chime choice is saved.':paused?'Chimes are paused. Your sound choice is saved.':reduced?'Reduced motion pauses automatic strikes. Your sound choice is saved.':playing?'Chimes ring only when the wind moves them.':recordingPending?'The recording is loading. Your sound choice is saved.':'Your sound choice is saved. Turn the speaker off and on to request audio again when the browser requires a tap.';
 }
 function unmount(){if(!mounted)return;mounted.panel.removeEventListener('input',mounted.input);mounted.panel.removeEventListener('click',mounted.click);mounted.panel.remove();mounted=null;}
 function refresh(){if(destroyed)return;const dialog=doc.getElementById('ohana-chat-dialog'),target=dialog?.open&&controller.session&&!doc.hidden?dialog.querySelector('.uc-command-sections'):null;if(mounted&&(!target||mounted.dialog!==dialog||!target.contains(mounted.panel)))unmount();if(target&&!mounted){const panel=doc.createElement('section');panel.className='ohana-chime-sound-controls';panel.dataset.chimeProtect='';panel.setAttribute('aria-label','Wind chime sound controls');panel.innerHTML='<h3>Wind chime sounds</h3><p>Separate settings on this device. Game sound mute still silences chimes.</p><div><span data-chime-status role="status" aria-live="polite"></span><label for="ohana-chime-volume">Wind chime volume</label><output for="ohana-chime-volume"></output><input id="ohana-chime-volume" aria-label="Wind chime volume" type="range" min="0" max="100" step="1"><button type="button" data-chime-mute><svg viewBox="0 0 48 48" focusable="false" aria-hidden="true"><path d="M5 18h10l11-9v30l-11-9H5z" fill="currentColor"/><path data-chime-waves d="M32 17c4 4 4 10 0 14M36 12c7 7 7 17 0 24" fill="none" stroke="currentColor" stroke-width="3" stroke-linecap="round"/><path data-chime-slash d="M5 43L43 5" fill="none" stroke="currentColor" stroke-width="4" stroke-linecap="round"/></svg></button><small></small></div>';const input=e=>{if(e.target.matches('input'))setVolume(Number(e.target.value)/100);},click=e=>{if(e.target.closest('[data-chime-mute]')){if(prefs.muted)void enable(e);else mute();}};panel.addEventListener('input',input);panel.addEventListener('click',click);target.append(panel);mounted={dialog,panel,input,click};}render();}
 const schedule=()=>{if(destroyed||queued)return;queued=true;queueMicrotask(()=>{queued=false;refresh();});};
 const observer=new MutationObserver(records=>{const relevant=records.some(r=>{if(r.type==='attributes')return r.target.id==='ohana-chat-dialog';if(r.target.closest?.('[data-daddy-audio-controls],.ohana-chime-sound-controls'))return false;if(mounted?.dialog.contains(r.target))return true;return[...r.addedNodes,...r.removedNodes].some(n=>n.nodeType===1&&(n.id==='ohana-chat-dialog'||n.querySelector?.('#ohana-chat-dialog')));});if(relevant)schedule();});observer.observe(doc.body,{childList:true,subtree:true,attributes:true,attributeFilter:['open']});
 const unsubscribe=controller.subscribe(s=>{const v=s.volume/.5;if(prefs.volume!==v){prefs.volume=v;save();}render();});
 function on(target,type,fn){target.addEventListener(type,fn);listeners.push(()=>target.removeEventListener(type,fn));}
 on(doc,'visibilitychange',()=>{if(doc.hidden)controller.mute();refresh();});on(events,'pagehide',()=>{controller.mute();unmount();});on(events,'storage',e=>{if(e.key===CHIME_SOUND_KEY||e.key===null){read();controller.setVolume(prefs.volume*.5);controller.mute();}if(e.key==='ohana_muted'||e.key===null)controller.globalMuteChanged();refresh();});on(doc,'click',()=>{controller.globalMuteChanged();render();});
 const style=doc.createElement('style');style.textContent='#ohana-chat-dialog .ohana-chime-sound-controls{display:block!important;max-width:100%;min-width:0;padding:12px;background:#fff5db;color:#173b35;border:1px solid #aa854f;border-radius:14px;margin:12px 0;text-shadow:none}.ohana-chime-sound-controls h3{font-size:18px}.ohana-chime-sound-controls p,.ohana-chime-sound-controls small{font-size:12px;line-height:1.45}.ohana-chime-sound-controls>div{display:grid;grid-template-columns:minmax(0,1fr) auto;gap:4px 8px}.ohana-chime-sound-controls input{grid-column:1;min-width:0;width:100%;min-height:44px}.ohana-chime-sound-controls button{grid-column:2;min-height:44px;min-width:70px}.ohana-chime-sound-controls small{grid-column:1/-1}.ohana-chime-sound-controls [data-chime-status]{position:absolute!important;width:1px!important;height:1px!important;padding:0!important;margin:-1px!important;overflow:hidden!important;clip:rect(0,0,0,0)!important;white-space:nowrap!important;border:0!important}.ohana-chime-sound-controls button[data-chime-mute]{grid-column:2;position:relative;min-width:48px!important;width:48px!important;min-height:48px!important;height:48px!important;padding:0!important;border:1px solid #aa854f!important;border-radius:12px!important;background:transparent!important;color:#173b35!important;box-shadow:none!important;display:grid;place-items:center;cursor:pointer;touch-action:manipulation}.ohana-chime-sound-controls button[data-chime-mute][data-silent=true]{color:#a22230!important;border-color:#c33b48!important}.ohana-chime-sound-controls button[data-chime-mute] svg{width:28px;height:28px;pointer-events:none}.ohana-chime-sound-controls button[data-chime-mute] [data-chime-slash]{stroke:#e42736;stroke-width:6;filter:none}.ohana-chime-sound-controls button[data-chime-mute][data-silent=false] [data-chime-waves]{stroke:#26795b}.ohana-chime-sound-controls button[data-chime-mute][data-silent=true] [data-chime-waves],.ohana-chime-sound-controls button[data-chime-mute][data-silent=false] [data-chime-slash]{display:none}.ohana-chime-sound-controls button[data-chime-mute]:focus-visible{outline:3px solid #0af0ee;outline-offset:3px}';doc.head.append(style);refresh();
 return{isMuted,wantsSound:()=>!prefs.muted,setVolume,mute,enable,refresh,get snapshot(){return{...prefs,enabled:controller.sound,available:controller.audio?.available===true,persisted,mounted:!!mounted};},destroy(){if(destroyed)return;destroyed=true;observer.disconnect();unsubscribe();listeners.splice(0).forEach(f=>f());if(controller.enableSound===enable)controller.enableSound=originalEnable;unmount();style.remove();}};
}
