export const PREF_KEY='ohana_chimes_device_v1';
const finite=(v,f=0)=>Number.isFinite(Number(v))?Number(v):f;
const clamp=(v,a,b)=>Math.max(a,Math.min(b,v));
export function readPrefs(storage){try{const p=JSON.parse(storage?.getItem(PREF_KEY)||'{}');return{volume:clamp(finite(p.volume,.18),0,.5),paused:p.paused===true};}catch{return{volume:.18,paused:false};}}

export class ChimeController{
 constructor({storage,random=Math.random,audio=null,now=()=>performance.now(),wallNow=Date.now,onChange=()=>{},allowSharedWeather=false}={}){
  Object.assign(this,{storage,random,audio,now,wallNow,onChange,allowSharedWeather});this.listeners=new Set();this.prefs=readPrefs(storage);this.visible=true;this.session=false;this.sessionKey=null;this.reduced=false;this.destroyed=false;this.sound=false;this.scene='';this.wind=this.simulated();this.rigs=[];this.last=null;this.nextStrike=Infinity;this.epoch=0;this.unlockIntent=0;this.stats={frames:0,contacts:0,strikes:0,discards:0};this.audioStatus=audio?.available?'Sound off':'Recording awaiting listening review';
 }
 simulated(){return{mode:'simulated',speedMps:2,gustMps:4,directionDeg:40};}
 get active(){return !this.destroyed&&this.visible&&this.session&&!this.prefs.paused;}
 get needsFrame(){return this.active&&!this.reduced&&this.rigs.length>0;}
 get snapshot(){return{scene:this.scene,mode:this.wind.mode,wind:{...this.wind},visible:this.visible,session:this.session,paused:this.prefs.paused,reduced:this.reduced,sound:this.sound,audioStatus:this.audioStatus,volume:this.prefs.volume,epoch:this.epoch,nextStrike:this.nextStrike,rigs:this.rigs.map(r=>({id:r.id,style:r.style,angle:r.angle,velocity:r.velocity,tubes:[...r.tubes],striker:r.striker,sail:r.sail})),stats:{...this.stats}};}
 notify(){const s=this.snapshot;this.onChange(s);for(const listener of this.listeners)listener(s);}
 subscribe(listener){if(this.destroyed)return()=>{};this.listeners.add(listener);return()=>this.listeners.delete(listener);}
 persist(){try{this.storage?.setItem(PREF_KEY,JSON.stringify(this.prefs));}catch{}this.notify();}
 schedule(t=this.now()){const strength=Math.max(.2,this.wind.speedMps+this.wind.gustMps*.2),gap=Math.max(2.8,12-strength*.65)*(.85+this.random()*.8);this.nextStrike=t+gap*1000;}
 reset(reason){if(this.destroyed)return;this.epoch++;this.audio?.stop(reason);this.last=null;this.nextStrike=Infinity;this.stats.discards++;if(this.active)this.schedule();this.notify();}
 setScene(id,specs=[]){if(this.destroyed)return;const key=JSON.stringify(specs.map(s=>typeof s==='string'?{id:s,style:'primary',count:5}:s));if(this.scene===id&&this.sceneSpec===key)return;this.scene=id;this.sceneSpec=key;this.rigs=specs.map((s,i)=>{if(typeof s==='string')s={id:s};const count=clamp(Math.floor(finite(s.count,5)),1,7);return{id:s.id,style:s.style||'primary',angle:0,velocity:0,phase:this.random()*Math.PI*2+i,tubes:Array(count).fill(0),tubeV:Array(count).fill(0),striker:0,strikerV:0,sail:0,sailV:0};});this.reset('scene-change');}
 setWind({speedMps=2,gustMps=4,directionDeg=40}={}){if(this.destroyed)return;this.wind={mode:'simulated',speedMps:clamp(finite(speedMps),0,15),gustMps:clamp(finite(gustMps),0,20),directionDeg:((finite(directionDeg)%360)+360)%360};this.notify();}
 setSharedWeather(state){if(this.destroyed)return false;const age=state&&this.wallNow()-state.updatedAt;if(!this.allowSharedWeather||!state||state.authorized!==true||state.available!==true||state.units!=='m/s'||!Number.isFinite(state.updatedAt)||!Number.isFinite(age)||age<0||age>1800000||!Number.isFinite(state.speedMps)||!Number.isFinite(state.gustMps)||!Number.isFinite(state.directionDeg)){this.setWind();return false;}this.wind={mode:'shared-weather',speedMps:clamp(state.speedMps,0,50),gustMps:clamp(state.gustMps,0,70),directionDeg:((state.directionDeg%360)+360)%360,updatedAt:state.updatedAt};this.notify();return true;}
 setVisible(v){if(this.destroyed)return;v=!!v;if(this.visible===v)return;this.visible=v;this.reset(v?'visible':'hidden');}
 setSession(key){if(this.destroyed)return;key=key===true?'explicit-session':key===false||key==null?null:String(key);if(this.sessionKey===key)return;this.sessionKey=key;this.session=key!==null;this.mute();this.wind=this.simulated();this.reset('session-change');}
 setReduced(v){if(this.destroyed)return;v=!!v;if(this.reduced===v)return;this.reduced=v;this.reset('reduced-motion');if(v)this.staticPose();}
 staticPose(){if(this.destroyed)return;for(const r of this.rigs){r.angle=r.velocity=r.striker=r.strikerV=r.sail=r.sailV=0;r.tubes.fill(0);r.tubeV.fill(0);}}
 setPaused(v){if(this.destroyed)return;this.prefs.paused=!!v;if(v)this.mute();this.reset('pause');this.persist();}
 setVolume(v){if(this.destroyed)return;this.prefs.volume=clamp(finite(v),0,.5);this.audio?.setVolume(this.prefs.volume);if(!this.prefs.volume)this.audio?.stop('zero-volume');this.persist();}
 hostMuted(){try{return!!this.audio?.isMuted?.();}catch{return true;}}
 hostAllowed(){try{return !this.hostMuted()&&(typeof this.audio?.allowed!=="function"||this.audio.allowed());}catch{return false;}}
 async enableSound(event){if(!event?.isTrusted||!this.active||!this.audio?.available)return false;const intent=++this.unlockIntent,epoch=this.epoch;try{const ok=await this.audio.unlock(event,this.prefs.volume);if(intent!==this.unlockIntent||epoch!==this.epoch||!this.active)return false;if(!this.hostAllowed()){this.mute();return false;}this.sound=!!ok;this.audioStatus=ok?'Sound enabled':'Sound unavailable';this.schedule();this.notify();return this.sound;}catch{if(intent!==this.unlockIntent||epoch!==this.epoch)return false;this.sound=false;this.audioStatus='Sound unavailable; remains silent';this.notify();return false;}}
 // A saved ON choice can resume only a recording that this page already unlocked with
 // a trusted gesture. A new/suspended context must wait for another real tap.
 resumeSoundIfUnlocked(){if(this.destroyed||!this.active||this.reduced||!this.audio?.available||this.prefs.volume<=0||this.sound)return false;
  if(this.audio.ctx?.state!=='running'||!this.audio.master||!this.audio.buffers?.length||!this.hostAllowed())return false;
  this.sound=true;this.audioStatus='Sound enabled';this.schedule();this.notify();return true;}
 mute(){if(this.destroyed)return;this.unlockIntent++;this.sound=false;this.audio?.stop('mute');this.audioStatus=this.audio?.available?'Sound off':'Recording awaiting listening review';this.notify();}
 globalMuteChanged(){if(this.destroyed)return;if(this.hostMuted())this.mute();}
 strike(event=null,{rigIndex=null,tubeIndex=null,contact=false}={}){if(this.hostMuted()){this.mute();return false;}if(!this.active||!this.sound||this.prefs.volume===0||event&&!event.isTrusted)return false;const r=this.rigs[rigIndex??Math.floor(this.random()*this.rigs.length)];if(!r)return false;const index=tubeIndex??Math.floor(this.random()*r.tubes.length);const played=this.audio.play({volume:this.prefs.volume,index,epoch:this.epoch});if(played){this.stats.strikes++;if(!this.reduced&&!contact)r.strikerV+=.07;this.notify();}return!!played;}
 step(t=this.now()){
  if(this.destroyed||!Number.isFinite(t))return this.rigs;if(this.sound&&this.hostMuted())this.mute();if(this.wind.mode==='shared-weather'&&(this.wallNow()-this.wind.updatedAt>1800000||this.wallNow()<this.wind.updatedAt))this.setWind();if(!this.active){this.last=null;return this.rigs;}if(this.last!==null&&(t<this.last||t-this.last>500)){this.reset('missed-frames');this.last=t;this.schedule(t);return this.rigs;}const dt=this.last===null?0:clamp((t-this.last)/1000,0,.05);this.last=t;this.stats.frames++;
  if(!this.reduced)for(const r of this.rigs){const wind=(Math.sin(t*.0007+r.phase)+.38*Math.sin(t*.00131+r.phase*2.3))*(this.wind.speedMps*.025+this.wind.gustMps*.01);r.velocity+=(-2.4*r.angle-1.15*r.velocity+wind)*dt;r.angle=clamp(r.angle+r.velocity*dt,-.085,.085);r.strikerV+=(-3.3*r.striker-1.05*r.strikerV+wind*1.2)*dt;r.striker=clamp(r.striker+r.strikerV*dt,-.115,.115);r.sailV+=(-1.7*r.sail-.95*r.sailV+wind*1.65)*dt;r.sail=clamp(r.sail+r.sailV*dt,-.14,.14);for(let i=0;i<r.tubes.length;i++){const k=3.4+i*.47,drive=wind*(.65+i*.055)+Math.sin(t*.0011+r.phase+i)*this.wind.speedMps*.007;r.tubeV[i]+=(-k*r.tubes[i]-1.1*r.tubeV[i]+drive)*dt;r.tubes[i]=clamp(r.tubes[i]+r.tubeV[i]*dt,-.11,.11);}}
  if(t>=this.nextStrike){if(!this.reduced&&(this.wind.speedMps>.2||this.wind.gustMps>.6)){let best=null;this.rigs.forEach((r,ri)=>r.tubes.forEach((a,ti)=>{const delta=Math.abs(a-r.striker);if(!best||delta>best.delta)best={r,ri,ti,delta};}));if(best&&best.delta>.012){best.r.tubeV[best.ti]*=-.35;best.r.strikerV*=-.35;this.stats.contacts++;this.strike(null,{rigIndex:best.ri,tubeIndex:best.ti,contact:true});}}this.schedule(t);}return this.rigs;
 }
 tick(t=this.now()){this.step(t);return this.snapshot;}
 destroy(){if(this.destroyed)return;this.destroyed=true;this.unlockIntent++;this.sound=false;this.audio?.destroy();this.rigs=[];this.last=null;this.nextStrike=Infinity;this.notify();this.listeners.clear();}
}
